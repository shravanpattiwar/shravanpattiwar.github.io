diabet=read.csv("G:/R Programs/Datasets/diabetes.csv")
View(diabet)
set.seed(2)
id=sample(2,nrow(diabet),prob=c(0.7,0.3),replace=TRUE)
diabet_train=diabet[id==1,]
diabet_test=diabet[id==2,]
Outcome=as.factor(diabet$Outcome)
colnames(diabet)
diabet_model=rpart(Outcome~.,data=diabet_train)
diabet_model
plot(diabet_model,margin=0.1)
text(diabet_model,use.n=TRUE,pretty=TRUE,cex=0.8)
pred_diabet=predict(diabet_model,newdata=diabet_test,type="vector")
pred_diabet
summary(diabet_model)
